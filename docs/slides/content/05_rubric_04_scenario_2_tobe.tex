  \begin{columns}[T]
    % Cột 1: 6 Bước tương tác mới số hóa
    \begin{column}{0.54\textwidth}
      \begin{block}{\footnotesize \textbf{6 Bước tương tác mới toàn diện (Quy trình To-Be)}}
        \begin{enumerate}
          \setlength\itemsep{0.8mm}
          \tiny
          \item \textbf{Chọn dịch vụ \& Khung giờ trực quan:} Tra cứu ma trận giờ trống thời gian thực; khóa giữ chỗ tức thì ($<$ 40 giây).
          \item \textbf{Xác nhận tức thì \& Gắn nhãn y tế đỏ:} Hệ thống tự nạp dặn dò dị ứng từ hồ sơ, cấp mã hẹn \& QR không cần chờ duyệt.
          \item \textbf{Bàn giao tiếp nhận QR an toàn 2 lớp:} Tiệm quét mã mở ngay cảnh báo đỏ; hai bên ký duyệt cam kết an toàn ($<$ 1 phút).
          \item \textbf{Live Tracking 4 mốc thời gian thực:} Thanh tiến trình cập nhật từng phút kèm Push Notification tự động từng giai đoạn.
          \item \textbf{Chủ động nhận thông báo \& Đón bé an toàn:} Nhận thông báo mời đến đón chuẩn xác; quét QR đối soát trả thú cưng.
          \item \textbf{Nhật ký số hóa \& Đặt lại lịch 1 chạm:} Lưu trữ trọn đời phác đồ dầu tắm, hóa đơn điện tử và nút Rebook nhanh.
        \end{enumerate}
      \end{block}
    \end{column}
