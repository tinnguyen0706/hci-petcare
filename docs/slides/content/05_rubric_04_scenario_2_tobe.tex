  \begin{columns}[T]
    % Cột 1: 6 Bước tương tác mới
    \begin{column}{0.50\textwidth}
      {\color{PetcarePrimary}\large \textbf{6 Bước tương tác số hóa (To-Be)}}\\[2mm]
      \small
      \begin{enumerate}
        \setlength\itemsep{1.8mm}
        \item \textbf{Chọn giờ trống:} Xem ma trận thời gian thực, khóa slot ($<$ 40s).
        \item \textbf{Xác nhận tức thì:} Tự nạp tiền sử dị ứng, cấp mã hẹn \& QR ngay.
        \item \textbf{Check-in 2 lớp:} Tiệm quét QR mở nhãn đỏ, ký cam kết an toàn.
        \item \textbf{Live Tracking:} Theo dõi 4 mốc thời gian thực kèm thông báo đẩy.
        \item \textbf{Thông báo đón bé:} Đến đón chuẩn xác theo tiến độ thực tế.
        \item \textbf{Nhật ký \& Rebook:} Lưu phác đồ sữa tắm trọn đời, đặt lại 1 chạm.
      \end{enumerate}
    \end{column}

    % Cột 2: Đột phá trải nghiệm so với As-Is dạng Bullet
    \begin{column}{0.46\textwidth}
      {\color{PetcarePrimary}\large \textbf{Đột phá trải nghiệm so với As-Is}}\\[2mm]
      \small
      \begin{itemize}
        \setlength\itemsep{2.2mm}
        \item \textbf{Tốc độ đặt lịch ($<$ 40 giây):} Khóa giờ trống trực tiếp trên app (thay vì chờ 15p--2h).
        \item \textbf{An toàn tuyệt đối (0\% rủi ro):} Tự nạp dị ứng, dán nhãn đỏ và ký cam kết trách nhiệm.
        \item \textbf{Tiếp nhận siêu tốc ($<$ 1 phút):} Quét QR tại quầy, xóa bỏ hoàn toàn sổ tay và giấy note.
        \item \textbf{Live Tracking thời gian thực:} 4 Mốc cập nhật minh bạch, an tâm làm việc khi gửi bé.
        \item \textbf{Nhật ký số hóa (1 chạm):} Lưu phác đồ dầu tắm trọn đời và tính năng Rebook tiện lợi.
      \end{itemize}
    \end{column}
  \end{columns}
