\begin{frame}{Giải Pháp Cốt Lõi}
  \framesubtitle{\color{white!90}\textbf{4 trụ cột trải nghiệm trên ứng dụng di động PetCare}}

  \begin{columns}[T]
    % Cột trái: 4 Trụ cột giải pháp
    \begin{column}{0.50\textwidth}
      {\color{PetcarePrimary}\large \textbf{4 Trụ cột trải nghiệm số hóa}}\\[2mm]
      \small
      \begin{itemize}
        \setlength\itemsep{2.8mm}
        \item \textbf{1. Đặt lịch tức thì ($<$ 40s):} Khóa slot giờ trống real-time, nhận mã hẹn \& QR check-in ngay lập tức.
        \item \textbf{2. Cảnh báo y tế (0\% rủi ro):} Tự động nạp tiền sử dị ứng, gắn nhãn đỏ cảnh báo, cam kết 2 lớp.
        \item \textbf{3. Live Tracking (4 mốc):} Theo dõi tiến độ thời gian thực minh bạch, nhận Push Notification tự động.
        \item \textbf{4. Nhật ký số hóa (1 chạm):} Lưu trọn đời phác đồ dầu tắm, hóa đơn điện tử \& Rebook nhanh chóng.
      \end{itemize}
    \end{column}

    % Cột phải: Mục tiêu trải nghiệm (UX Goals)
    \begin{column}{0.46\textwidth}
      {\color{PetcarePrimary}\large \textbf{Mục tiêu trải nghiệm (UX Goals)}}\\[2mm]
      \small
      \begin{itemize}
        \setlength\itemsep{2.8mm}
        \item \textbf{Tốc độ vượt trội:} Rút ngắn thời gian thao tác đặt lịch từ 15--120p xuống dưới \textbf{40 giây}.
        \item \textbf{An toàn tuyệt đối:} Triệt tiêu \textbf{100\%} rủi ro dùng nhầm sữa tắm do quên dặn dò khi đổi ca.
        \item \textbf{Minh bạch tiến độ:} Xóa bỏ hoàn toàn khoảng mù thông tin trong suốt \textbf{2--3 tiếng} chăm sóc.
        \item \textbf{Mức độ hài lòng:} Hướng đến điểm Usability \textbf{SUS $>$ 80} (thang đo chuẩn quốc tế).
      \end{itemize}
    \end{column}
  \end{columns}
\end{frame}
