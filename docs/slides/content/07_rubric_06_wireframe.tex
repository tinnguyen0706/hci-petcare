        \tiny
        \begin{itemize}
          \setlength\itemsep{0.5mm}
          \item \textbf{Ideal State:} Ma trận giờ trực quan, dữ liệu hiển thị hoàn chỉnh.
          \item \textbf{Loading State:} Skeleton Loading chống giật layout khi tải.
          \item \textbf{Empty State:} Minh họa thân thiện kèm nút CTA bắt đầu đặt lịch.
          \item \textbf{Error/Conflict State:} Cảnh báo dị ứng \& mã QR cứu hộ offline.
          \item \textbf{Success State:} Chúc mừng kèm mã đặt lịch và QR tiếp nhận.
        \end{itemize}
      \end{exampleblock}
