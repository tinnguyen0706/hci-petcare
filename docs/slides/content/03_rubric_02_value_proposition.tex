  \framesubtitle{\color{PetcareSecondary}\textbf{[Rubric: 10\%] Đầy đủ các phần Canvas $\bullet$ Đối ứng trực tiếp 1--1 giữa Persona và Giá trị đề xuất}}

  \begin{columns}[T]
    % Cột 1: Đối ứng Persona 1 (Lan)
    \begin{column}{0.49\textwidth}
      \begin{block}{\footnotesize \textbf{Bản đồ Giá trị -- Persona 1 (Nguyễn Hoàng Lan)}}
        \tiny
        \textbf{Hồ sơ khách hàng (Customer Profile):}
        \begin{itemize}
          \setlength\itemsep{0.5mm}
          \item \textit{Nhiệm vụ (Jobs):} Đặt spa định kỳ, bảo vệ cún dị ứng xà phòng.
          \item \textit{Điểm đau (Pains):} Chờ chat 15p--2h; rủi ro quên dị ứng; mù mờ tiến độ.
          \item \textit{Kỳ vọng (Gains):} Xác nhận tức thì, thợ nhớ dị ứng, cập nhật từ xa.
        \end{itemize}
        \vspace{0.5mm}
        \hrule
        \vspace{0.5mm}
        \textbf{Bản đồ giải pháp (Value Map) -- Đối ứng 1--1:}
        \begin{itemize}
          \setlength\itemsep{0.5mm}
          \item \textbf{Khóa giờ trống tức thì} $\leftrightarrow$ Xóa bỏ 100\% thời gian chờ đợi.
          \item \textbf{Tự động dán nhãn y tế đỏ} $\leftrightarrow$ Triệt tiêu nguy cơ tiệm quên dị ứng.
          \item \textbf{Live Tracking 4 mốc} $\leftrightarrow$ Xóa điểm mù thông tin 2--3 tiếng.
          \item \textbf{Nhật ký số hóa \& Rebook} $\leftrightarrow$ Lưu phác đồ sữa tắm, đặt lại 1 chạm.
        \end{itemize}
      \end{block}
    \end{column}

    % Cột 2: Đối ứng Persona 2 (Khoa)
    \begin{column}{0.49\textwidth}
      \begin{exampleblock}{\footnotesize \textbf{Bản đồ Giá trị -- Persona 2 (Trần Minh Khoa)}}
        \tiny
        \textbf{Hồ sơ khách hàng (Customer Profile):}
        \begin{itemize}
          \setlength\itemsep{0.5mm}
          \item \textit{Nhiệm vụ (Jobs):} Tìm dịch vụ spa an toàn cho mèo nhút nhát; quản lý phí.
          \item \textit{Điểm đau (Pains):} Sợ mèo hoảng loạn vì đông; sợ sự cố; ngại gọi điện thoại.
          \item \textit{Kỳ vọng (Gains):} Đặt giờ vắng, theo dõi an toàn từ xa, giá minh bạch.
        \end{itemize}
        \vspace{0.5mm}
        \hrule
        \vspace{0.5mm}
        \textbf{Bản đồ giải pháp (Value Map) -- Đối ứng 1--1:}
        \begin{itemize}
          \setlength\itemsep{0.5mm}
          \item \textbf{Ma trận hiển thị giờ vắng} $\leftrightarrow$ Chủ động đặt khung giờ yên tĩnh.
          \item \textbf{Buồng cách ly \& Thông báo đón} $\leftrightarrow$ Tránh mèo tiếp xúc chó dữ.
          \item \textbf{Cập nhật trạng thái tự động} $\leftrightarrow$ An tâm tuyệt đối mà không cần gọi hỏi.
          \item \textbf{Bảng giá \& Hóa đơn điện tử} $\leftrightarrow$ Minh bạch 100\%, không phát sinh phí.
        \end{itemize}
      \end{exampleblock}
    \end{column}
  \end{columns}
