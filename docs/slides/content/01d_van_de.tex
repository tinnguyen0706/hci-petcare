\begin{frame}{4 Điểm Đau Cốt Lõi}
  \framesubtitle{\color{white!90}\textbf{Đứt gãy trải nghiệm trong quy trình chăm sóc thủ công}}

  \begin{columns}[T]
    % Cột trái: 4 Điểm đau
    \begin{column}{0.52\textwidth}
      {\color{PetcarePrimary}\large \textbf{4 Rào cản trải nghiệm cốt lõi}}\\[2mm]
      \small
      \begin{itemize}
        \setlength\itemsep{2.8mm}
        \item \textbf{1. Chờ đợi bất định:} Nhắn tin/gọi điện hỏi giờ, chờ 15p--2h; dễ bị trùng hoặc lỡ lịch.
        \item \textbf{2. Rủi ro y tế sinh mệnh:} Tiền sử dị ứng xà phòng dặn miệng dễ bị quên khi nhân viên đổi ca.
        \item \textbf{3. Điểm mù tiến độ:} Khoảng trống thông tin 2--3 tiếng gây bất an nhưng ngại gọi chen ngang.
        \item \textbf{4. Mất dấu lịch sử:} Không lưu trữ phác đồ dầu tắm, khó chăm sóc sức khỏe lâu dài.
      \end{itemize}
    \end{column}

    % Cột phải: Hậu quả & Tổn thất thực tế dạng Bullet
    \begin{column}{0.44\textwidth}
      {\color{PetcarePrimary}\large \textbf{Hậu quả \& Tổn thất thực tế}}\\[2mm]
      \small
      \begin{itemize}
        \setlength\itemsep{3mm}
        \item \textbf{Chờ xác nhận kéo dài (15 -- 120p):} Trì trệ lịch trình cá nhân, dễ bị mất slot trống vào tay khách khác.
        \item \textbf{Nguy cơ quên dặn dị ứng ($>$ 80\%):} Giao ca bằng sổ tay/dặn miệng làm tái phát viêm da cún cưng.
        \item \textbf{Điểm mù tiến độ (2 -- 3 tiếng):} Khách hàng rơi vào trạng thái bồn chồn, bất an suốt buổi họp.
      \end{itemize}
    \end{column}
  \end{columns}
\end{frame}
