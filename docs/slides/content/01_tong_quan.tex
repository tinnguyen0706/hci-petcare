\begin{frame}{Nghịch lý thời gian: Khi 83.3\% chủ nuôi bận rộn đối mặt quy trình thủ công}
  \begin{columns}[T]
    \begin{column}{0.58\textwidth}
      \textbf{4 Điểm nghẽn nghiêm trọng trong quy trình chăm sóc cũ:}
      \begin{itemize}
        \item \textbf{Chờ đợi phản hồi thụ động:} Mất từ 15--45 phút nhắn tin/gọi điện hỏi lịch, dễ bị trùng hoặc lỡ mất khung giờ mong muốn.
        \item \textbf{Dặn dò dị ứng bị lãng quên:} Tiền sử viêm da dị ứng xà phòng dễ bị thất lạc khi chuyển ca từ lễ tân sang thợ tắm.
        \item \textbf{``Hố đen thông tin'' suốt 2--3 giờ:} Hoàn toàn mù mờ về tiến độ chăm sóc; bất an nhưng ngại gọi điện làm phiền.
        \item \textbf{Dữ liệu phân mảnh:} Không có nơi lưu trữ nhật ký sản phẩm đã dùng, đơn giá và tình trạng da lông qua từng tháng.
      \end{itemize}
    \end{column}

    \begin{column}{0.40\textwidth}
      \begin{alertblock}{Hệ quả thực tế}
        Chủ nuôi liên tục ở trạng thái \textbf{bị động, bất an và lãng phí thời gian} trao đổi lặp đi lặp lại.
      \end{alertblock}
      \vskip 2mm
      \begin{exampleblock}{Tiếng nói người dùng}
        \footnotesize \textit{``Mình rất bận nên chỉ mong đặt lịch nhanh có xác nhận ngay, gửi bé đi có thể theo dõi tiến độ từ xa và tiệm luôn nhớ bé bị dị ứng da để tắm an toàn.''}
        \vskip 1mm
        \hfill --- \textbf{Chị Lan (28 tuổi, TP.HCM)}
      \end{exampleblock}
    \end{column}
  \end{columns}
\end{frame}
