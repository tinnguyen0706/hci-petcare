      \begin{block}{\footnotesize \textbf{4 Đoạn hành trình đầy khó khăn của quy trình cũ}}
        \begin{itemize}
          \setlength\itemsep{1.2mm}
          \tiny
          \item \textbf{1. Đặt lịch \& Chờ đợi bất định:} Gọi điện hoặc nhắn tin hỏi giờ; nhân viên tra sổ tay; chờ 15p--2h; thường xuyên bị người khác đặt mất chỗ.
          \item \textbf{2. Dặn dò dị ứng rời rạc:} Tự nhắc miệng hoặc gõ chat; nhân viên dán giấy post-it dễ rơi; đổi ca trực bỏ sót ($>$ 80\% rủi ro viêm da).
          \item \textbf{3. Điểm mù tiến độ (Black-box):} Thú cưng ở tiệm 2--3 tiếng; chủ nuôi nơm nớp lo âu tại nơi làm việc nhưng ngại gọi điện làm phiền tiệm.
          \item \textbf{4. Đón bé \& Thất lạc lịch sử:} Nhận cuộc gọi gián đoạn công việc; hóa đơn giấy dễ vứt bỏ; không nhớ loại sữa tắm trị liệu lần trước.
        \end{itemize}
      \end{block}
    \end{column}
