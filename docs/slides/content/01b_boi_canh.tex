\begin{frame}{Bối Cảnh Đề Tài}
  \framesubtitle{\color{white!90}\textbf{Nghịch lý thời gian của chủ nuôi thú cưng đô thị}}

  \begin{columns}[T]
    % Cột 1: Xu hướng & Chủ nuôi
    \begin{column}{0.48\textwidth}
      {\color{PetcarePrimary}\large \textbf{Xu hướng \& Khách hàng đô thị}}\\[2mm]
      \small
      \begin{itemize}
        \setlength\itemsep{2.8mm}
        \item \textbf{Nhân hóa thú cưng:} Coi như con cái trong gia đình, nhu cầu tắm spa định kỳ tăng mạnh.
        \item \textbf{Quỹ thời gian eo hẹp:} Chủ nuôi bận rộn với công việc, cần các dịch vụ nhanh gọn.
        \item \textbf{Tâm lý bất an:} Luôn lo lắng về rủi ro dị ứng và an toàn của thú cưng khi gửi tiệm.
      \end{itemize}
    \end{column}

    % Cột 2: Thực trạng spa hiện nay
    \begin{column}{0.48\textwidth}
      {\color{PetcarePrimary}\large \textbf{Thực trạng quy trình thủ công}}\\[2mm]
      \small
      \begin{itemize}
        \setlength\itemsep{2.8mm}
        \item \textbf{Kênh liên hệ rời rạc:} Nhắn tin mạng xã hội, gọi điện thoại dễ trôi dạt thông tin.
        \item \textbf{Đối soát lịch thủ công:} Nhân viên tra sổ tay tìm giờ trống, dễ bị trùng lịch hẹn.
        \item \textbf{Ghi nhớ cảm tính:} Dán giấy note lên lồng, nguy cơ cao khi bàn giao đổi ca trực.
      \end{itemize}
    \end{column}
  \end{columns}

  \vspace{4mm}
  \begin{center}
    \small
    \textbf{Nghịch lý cốt lõi:} Nhu cầu chăm sóc tăng cao nhưng quy trình tương tác vẫn \textbf{thủ công, phân mảnh và phụ thuộc trí nhớ cá nhân}.
  \end{center}
\end{frame}
